cmake_minimum_required(VERSION 3.16)
project(object_tracking_cpp)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(Python3 3.12 EXACT REQUIRED COMPONENTS Interpreter Development)
find_package(pybind11 CONFIG REQUIRED)
add_executable(object_follow_cpp src/object_follow.cpp)
target_compile_options(object_follow_cpp PRIVATE -Wall -Wextra -Wpedantic)
set_target_properties(object_follow_cpp PROPERTIES CXX_VISIBILITY_PRESET hidden)
ament_target_dependencies(object_follow_cpp rclcpp geometry_msgs sensor_msgs std_msgs std_srvs tf2 tf2_ros)
target_link_libraries(object_follow_cpp pybind11::embed Python3::Python)
install(TARGETS object_follow_cpp DESTINATION lib/${PROJECT_NAME})
ament_package()
